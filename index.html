const SHEET_PARELLES = 'PARELLES'; const SHEET_PARTITS = 'PARTITS'; const SHEET_DISPONIBILITAT = 'DISPONIBILITAT'; function doGet(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const params = e && e.parameter ? e.parameter : {}; const action = String(params.action || '').trim(); let resultat; if (action === 'save_day') { const items = params.items ? JSON.parse(params.items) : []; resultat = saveDay_(ss, { id_parella: params.id_parella, parella: params.parella, data: params.data, dia: params.dia, items }); } else { resultat = { ok: true, parelles: getParelles_(ss), partits: getPartits_(ss), disponibilitat: getDisponibilitat_(ss) }; } const json = JSON.stringify(resultat); const callback = params.callback ? String(params.callback).trim() : ''; if (callback) { if (!/^[A-Za-z_$][A-Za-z0-9_$]*$/.test(callback)) { throw new Error('Callback no vàlid'); } return ContentService .createTextOutput(`${callback}(${json});`) .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService .createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); } catch (error) { const json = JSON.stringify({ ok: false, error: error.message }); const callback = e && e.parameter ? String(e.parameter.callback || '').trim() : ''; if (callback && /^[A-Za-z_$][A-Za-z0-9_$]*$/.test(callback)) { return ContentService .createTextOutput(`${callback}(${json});`) .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService .createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); } } function saveDay_(ss, dades) { const lock = LockService.getScriptLock(); lock.waitLock(30000); try { const sheet = ss.getSheetByName(SHEET_DISPONIBILITAT); if (!sheet) throw new Error('No existeix la pestanya DISPONIBILITAT'); const idParella = String(dades.id_parella || '').trim(); const parella = String(dades.parella || '').trim(); const data = String(dades.data || '').trim(); const dia = String(dades.dia || '').trim(); const items = Array.isArray(dades.items) ? dades.items : []; if (!idParella || !data) throw new Error('Falten dades obligatòries'); if (!/^\d{4}-\d{2}-\d{2}$/.test(data)) throw new Error('Data no vàlida'); const lastRow = sheet.getLastRow(); if (lastRow >= 2) { const values = sheet.getRange(2, 1, lastRow - 1, 7).getValues(); const rowsToDelete = []; for (let i = 0; i < values.length; i++) { if (String(values[i][0] || '').trim() === idParella && formatDate_(values[i][2]) === data) { rowsToDelete.push(i + 2); } } for (let i = rowsToDelete.length - 1; i >= 0; i--) { sheet.deleteRow(rowsToDelete[i]); } } const newRows = []; items.forEach(item => { const franja = String(item.franja || '').trim(); if (!franja) return; newRows.push([ idParella, parella, data, dia, franja, String(item.hora_altra || '').trim(), new Date() ]); }); if (newRows.length) { const startRow = sheet.getLastRow() + 1; sheet.getRange(startRow, 1, newRows.length, 7).setValues(newRows); } SpreadsheetApp.flush(); return { ok: true, savedItems: newRows.length, data }; } finally { lock.releaseLock(); } } function doPost(e) { try { // La nova versió envia un formulari amb el JSON dins de "payload". // També mantenim compatibilitat amb el POST JSON anterior. let dades = null; if (e && e.parameter && e.parameter.payload) { dades = JSON.parse(e.parameter.payload); } else if (e && e.postData && e.postData.contents) { dades = JSON.parse(e.postData.contents); } else { throw new Error('No s’han rebut dades'); } const ss = SpreadsheetApp.getActiveSpreadsheet(); // Nova sincronització mensual. if (dades.action === 'sync_month') { return syncMonth_(ss, dades); } // Compatibilitat amb el POST anterior d'un sol dia. return saveSingleAvailability_(ss, dades); } catch (error) { return jsonResponse_({ ok: false, error: error.message }); } } function syncMonth_(ss, dades) { const lock = LockService.getScriptLock(); lock.waitLock(30000); try { const sheet = ss.getSheetByName(SHEET_DISPONIBILITAT); if (!sheet) { throw new Error('No existeix la pestanya DISPONIBILITAT'); } const idParella = String(dades.id_parella || '').trim(); const parella = String(dades.parella || '').trim(); const month = String(dades.month || '').trim(); const items = Array.isArray(dades.items) ? dades.items : []; if (!idParella || !month) { throw new Error('Falten dades obligatòries'); } if (!/^\d{4}-\d{2}$/.test(month)) { throw new Error('Mes no vàlid'); } // IMPORTANT: no fem deleteRow() una per una. // Això podia fer que un guardat trigues massa i que la web donés // per acabada l'operació abans que Google Sheets hagués acabat. // Llegim totes les files, eliminem en memòria les de la parella/mes // i escrivim el resultat en un sol bloc. const lastRow = sheet.getLastRow(); const existing = lastRow >= 2 ? sheet.getRange(2, 1, lastRow - 1, 7).getValues() : []; const remaining = existing.filter(row => { const existingId = String(row[0] || '').trim(); const existingDate = formatDate_(row[2]); return !( existingId === idParella && existingDate.startsWith(month) ); }); // Preparem les noves files del mes. const newRows = []; items.forEach(item => { const data = String(item.data || '').trim(); const dia = String(item.dia || '').trim(); const franja = String(item.franja || '').trim(); const horaAltra = String(item.hora_altra || '').trim(); if (!data || !franja) return; if (!data.startsWith(month)) return; newRows.push([ idParella, parella, data, dia, franja, horaAltra, new Date() ]); }); const allRows = remaining.concat(newRows); const oldRowCount = Math.max(lastRow - 1, 0); if (oldRowCount > 0) { sheet.getRange(2, 1, oldRowCount, 7).clearContent(); } if (allRows.length > 0) { sheet.getRange(2, 1, allRows.length, 7).setValues(allRows); } SpreadsheetApp.flush(); return jsonResponse_({ ok: true, message: 'Disponibilitat mensual actualitzada', savedItems: newRows.length }); } finally { lock.releaseLock(); } } function saveSingleAvailability_(ss, dades) { const sheet = ss.getSheetByName(SHEET_DISPONIBILITAT); if (!sheet) { throw new Error('No existeix la pestanya DISPONIBILITAT'); } const idParella = String(dades.id_parella || '').trim(); const parella = String(dades.parella || '').trim(); const data = String(dades.data || '').trim(); const dia = String(dades.dia || '').trim(); const franja = String(dades.franja || '').trim(); const horaAltra = String(dades.hora_altra || '').trim(); if (!idParella || !data || !franja) { throw new Error('Falten dades obligatòries'); } const lastRow = sheet.getLastRow(); if (lastRow >= 2) { const values = sheet .getRange(2, 1, lastRow - 1, 7) .getValues(); for (let i = values.length - 1; i >= 0; i--) { const existingId = String(values[i][0] || '').trim(); const existingDate = formatDate_(values[i][2]); if ( existingId === idParella && existingDate === data ) { sheet.deleteRow(i + 2); } } } sheet.appendRow([ idParella, parella, data, dia, franja, horaAltra, new Date() ]); return jsonResponse_({ ok: true, message: 'Disponibilitat guardada' }); } function getParelles_(ss) { const sheet = ss.getSheetByName(SHEET_PARELLES); if (!sheet) { throw new Error('No existeix la pestanya PARELLES'); } const values = sheet.getDataRange().getValues(); const resultat = []; for (let i = 1; i < values.length; i++) { if (!values[i][0]) continue; resultat.push({ id: String(values[i][0]), name: String(values[i][1]), category: String(values[i][2]) }); } return resultat; } function getPartits_(ss) { const sheet = ss.getSheetByName(SHEET_PARTITS); if (!sheet) { throw new Error('No existeix la pestanya PARTITS'); } const values = sheet.getDataRange().getValues(); const resultat = []; for (let i = 1; i < values.length; i++) { if (!values[i][0]) continue; resultat.push({ id: String(values[i][0]), category: String(values[i][1]), pairAId: String(values[i][2]), pairA: String(values[i][3]), pairBId: String(values[i][4]), pairB: String(values[i][5]), status: String(values[i][6]), date: formatDate_(values[i][7]), hour: formatTime_(values[i][8]) }); } return resultat; } function getDisponibilitat_(ss) { const sheet = ss.getSheetByName(SHEET_DISPONIBILITAT); if (!sheet) { throw new Error('No existeix la pestanya DISPONIBILITAT'); } const values = sheet.getDataRange().getValues(); const resultat = []; for (let i = 1; i < values.length; i++) { if (!values[i][0]) continue; resultat.push({ pairId: String(values[i][0]), pair: String(values[i][1]), date: formatDate_(values[i][2]), day: String(values[i][3] || ''), slot: String(values[i][4] || ''), otherHour: String(values[i][5] || '') }); } return resultat; } function formatDate_(value) { if (!value) return ''; if (Object.prototype.toString.call(value) === '[object Date]') { return Utilities.formatDate( value, Session.getScriptTimeZone(), 'yyyy-MM-dd' ); } return String(value).substring(0, 10); } function formatTime_(value) { if (!value) return ''; if (Object.prototype.toString.call(value) === '[object Date]') { return Utilities.formatDate( value, Session.getScriptTimeZone(), 'HH:mm' ); } return String(value); } function jsonResponse_(object) { return ContentService .createTextOutput(JSON.stringify(object)) .setMimeType(ContentService.MimeType.JSON); }
